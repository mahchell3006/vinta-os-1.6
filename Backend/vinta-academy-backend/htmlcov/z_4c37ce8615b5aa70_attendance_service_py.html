13def check_in_student( 

14 session_id: str, student_id: str, academy_id: str, checked_in_by: str, 

15 status: str = "PRESENT", is_group_swap: bool = False, 

16 apply_billing: bool = True, 

17) -> SessionStudent: 

18 """Mark a student as checked in to a session. 

19 

20 Args: 

21 session_id: session being attended. 

22 student_id: student attending. 

23 academy_id: tenant scope (also used for billing side effects). 

24 checked_in_by: staff user id recording the check-in — PIN 

25 attribution (``@verify_staff_pin``) is enforced at the route 

26 layer; the id is stored on ``checked_in_by``. 

27 status: PRESENT | ABSENT. Defaults to PRESENT, because this is the 

28 *check-in* operation — the caller is asserting the student is 

29 here. (The blank-slate ABSENT rows that a new roster starts from 

30 are created by ``materialize_roster`` / ``add_student_to_session``, 

31 not here.) Whether an absence still consumes a credit is the 

32 academy's ``absence_consumes_credit`` toggle; that decision is 

33 made when the class is finalised, not at check-in. An 

34 unrecognised value raises rather than guessing, since guessing 

35 PRESENT would charge money the caller never asked to charge. 

36 is_group_swap: True for guest check-ins into a different group — 

37 billing resolves the level-based subscription. 

38 apply_billing: run billing side effects inline (default True). 

39 Pass False when the caller drives 

40 ``billing_service.record_checkin_billing_side_effects`` itself 

41 to avoid double-charging. 

42 

43 After the record is created/updated, billing side effects run via 

44 ``billing_service.record_checkin_billing_side_effects`` (imported 

45 lazily to avoid a circular import). 

46 """ 

47 status = (status or "PRESENT").upper() 

48 if status not in ("PRESENT", "ABSENT"): 

49 raise ValueError(f"status must be PRESENT or ABSENT, got {status!r}") 

50 

51 record = SessionStudent.query.filter_by( 

52 session_id=session_id, student_id=student_id 

53 ).first() 

54 

55 now = datetime.now(timezone.utc) 

56 if record: 

57 # Update existing record 

58 record.is_present = (status == "PRESENT") 

59 record.status = status 

60 record.is_group_swap = bool(is_group_swap) 

61 record.timestamp = now 

62 if status == "PRESENT": 

63 record.checked_in_at = now 

64 record.checked_in_by = checked_in_by 

65 else: 

66 # Create new record 

67 record = SessionStudent( 

68 id=str(uuid.uuid4()), 

69 session_id=session_id, 

70 student_id=student_id, 

71 is_present=(status == "PRESENT"), 

72 status=status, 

73 is_group_swap=bool(is_group_swap), 

74 timestamp=now, 

75 checked_in_at=now if status == "PRESENT" else None, 

76 checked_in_by=checked_in_by, 

77 ) 

78 db.session.add(record) 

79 

80 # Audit log 

81 from app.models.student import Student 

82 student = db.session.get(Student, student_id) 

83 student_name = f"{student.first_name} {student.last_name}" if student else "Unknown" 

84 

85 log = ActivityLog( 

86 id=str(uuid.uuid4()), 

87 academy_id=academy_id, 

88 # None, never the string "system": user_id is a foreign key to 

89 # users.id and SQLite enforces it, so a sentinel that is not a real 

90 # user raises IntegrityError. NULL means "no human did this" and the 

91 # activity-log reader renders it as "System". 

92 user_id=checked_in_by or None, 

93 entity_type="session", 

94 entity_id=session_id, 

95 action="checked_in", 

96 description=f"Student check-in — {student_name} ({status})", 

97 ) 

98 db.session.add(log) 

99 db.session.flush() 

100 

101 if apply_billing: 

102 from app.services.billing_service import ( 

103 record_checkin_billing_side_effects as _billing_side_effects, 

104 ) 

105 _billing_side_effects( 

106 session_id=session_id, 

107 student_id=student_id, 

108 status=status, 

109 is_group_swap=bool(is_group_swap), 

110 checked_in_by=checked_in_by, 

111 academy_id=academy_id, 

112 ) 

113 

114 return record

118 session_id: str, student_id: str, academy_id: str 

119) -> SessionStudent | None: 

120 """Check out a student from a session.""" 

121 record = SessionStudent.query.filter_by( 

122 session_id=session_id, student_id=student_id 

123 ).first() 

124 

125 if not record or not record.is_present: 

126 return None 

127 

128 record.checked_out_at = datetime.now(timezone.utc) 

129

145 )

152def auto_checkout_session( 

153 session_id: str, academy_id: str, staff_id: str | None = None 

154) -> int: 

155 """ 

156 Auto check-out all present students when session ends. 

157 Returns count of students checked out. 

158 

159 ``staff_id`` is the person who caused this — the staff member who ended 

160 the class, or the one who hit the auto-checkout route. An automatic 

161 check-out is still attributable to whoever triggered it, so it is logged 

162 against them; it falls back to NULL (rendered as "System") only when 

163 nothing human triggered it, as in the cron sweep. 

164 """ 

165 now = datetime.now(timezone.utc) 

166 records = SessionStudent.query.filter_by( 

167 session_id=session_id, is_present=True 

168 ).filter(SessionStudent.checked_out_at.is_(None)).all() 

169 

170 count = 0 

171 for record in records: 

172 record.checked_out_at = now 

173 count += 1 

174 

175 if count > 0: 

176 log = ActivityLog( 

177 id=str(uuid.uuid4()), 

178 academy_id=academy_id, 

179 # NULL, never the string "system" — user_id is a foreign key to 

180 # users.id and SQLite enforces it. The sentinel raised 

181 # IntegrityError here on every call that had anyone to check out, 

182 # which is why this path never once succeeded. 

183 user_id=staff_id or None, 

184 entity_type="session", 

185 entity_id=session_id, 

186 action="checked_out", 

187 description=f"Auto check-out — {count} students", 

188 ) 

189 db.session.add(log) 

190 db.session.flush() 

191 

192 return count

195def get_session_roster(session_id: str) -> list: 

196 """Get the full roster for a session with attendance status.""" 

197 records = SessionStudent.query.filter_by(session_id=session_id).all() 

198 

199 roster = [] 

200 for record in records: 

201 student = record.student 

202 roster.append({ 

203 "id": record.id, 

204 "student_id": record.student_id, 

205 "student_name": f"{student.first_name} {student.last_name}" if student else "Unknown", 

206 "is_present": record.is_present, 

207 "status": record.status, 

208 "is_group_swap": record.is_group_swap, 

209 "checked_in_at": record.checked_in_at.isoformat() if record.checked_in_at else None, 

210 "checked_out_at": record.checked_out_at.isoformat() if record.checked_out_at else None, 

211 "checked_in_by": record.checked_in_by, 

212 "timestamp": record.timestamp.isoformat() if record.timestamp else None, 

213 }) 

214 

215 return roster

218def get_session_roster_with_badges(session_id: str) -> list: 

219 """ 

220 Roster enriched with subscription status per student: 

221 remaining_credits / access_end plus RENEW_REQUIRED / 

222 ATTENDANCE_WARNING badges. Purely additive — base roster untouched. 

223 """ 

224 from datetime import date 

225 from app.models.billing import StudentSubscription 

226 from app.models.class_room import Class 

227 

228 roster = get_session_roster(session_id) 

229 if not roster: 

230 return roster 

231 

232 session = db.session.get(Session, session_id) 

233 group = db.session.get(Class, session.class_id) if session else None 

234 

235 for entry in roster: 

236 student_id = entry.get("student_id") 

237 subs = StudentSubscription.query.filter_by( 

238 student_id=student_id, status="ACTIVE" 

239 ).all() 

240 

241 # Prefer the subscription for this session's group 

242 sub = next((s for s in subs if session and s.group_id == session.class_id), None) 

243 

244 remaining = sub.remaining_credits if sub else None 

245 access_end = None 

246 for s in subs: 

247 if s.access_end_date and (access_end is None or s.access_end_date > access_end): 

248 access_end = s.access_end_date 

249 

250 entry["remaining_credits"] = remaining 

251 entry["access_end"] = access_end.isoformat() if access_end else None 

252 

253 badges = [] 

254 if not subs: 

255 badges.append("RENEW_REQUIRED") 

256 else: 

257 if sub and sub.billing_model == "CREDIT_BASED" and (sub.remaining_credits or 0) <= 0: 

258 badges.append("RENEW_REQUIRED") 

259 if access_end and access_end < date.today(): 

260 badges.append("RENEW_REQUIRED") 

261 

262 if group and group.enforce_attendance and group.attendance_threshold: 

263 total = SessionStudent.query.filter_by(student_id=student_id).join( 

264 Session, SessionStudent.session_id == Session.id 

265 ).filter(Session.class_id == group.id).count() 

266 attended = SessionStudent.query.filter_by( 

267 student_id=student_id, is_present=True 

268 ).join(Session, SessionStudent.session_id == Session.id).filter( 

269 Session.class_id == group.id 

270 ).count() 

271 # The value is stored under two conventions: a fraction (0.75, the 

272 # model default) and a percentage (75, what the class form writes). 

273 # Normalise before comparing — read raw, `ratio < 75` is true for 

274 # every student and the warning never clears. 

275 threshold = float(group.attendance_threshold) 

276 if threshold > 1: 

277 threshold = threshold / 100.0 

278 if total > 0 and (attended / total) < threshold: 

279 badges.append("ATTENDANCE_WARNING") 

280 

281 entry["badges"] = badges 

282 

283 return roster
