17def auto_checkout_expired_sessions(): 

18 """ 

19 Check all in-progress sessions whose end_time has passed and auto check-out 

20 any students still marked present. 

21 

22 Deliberately does NOT end the class. 

23 

24 This job used to set ``session.status = "completed"`` the moment the clock 

25 passed ``end_time``. That is the one thing it must never do. Ending a class 

26 is what settles money — ``session_lifecycle_service.end_session`` runs 

27 ``settle_absences``, writes ``conducted``, and stamps ``actual_end_time`` 

28 and ``ended_by_staff_id`` behind a staff PIN. A timer cannot consent to that, 

29 and the terminal status also drops the session out of the 

30 ``status IN ('scheduled','in_progress')`` query the Classrooms tab uses, so 

31 the desk lost the Finish / Extend / Void choices entirely. 

32 

33 So the class is left IN_PROGRESS and the desk is told it is over (the 

34 client derives "overdue" from ``end_time`` and raises the prompt). This job 

35 only tidies the register: a student left checked in overnight should not 

36 stay checked in forever. 

37 

38 Two bugs used to stop this function before it did anything at all: 

39 

40 * ``import auto_checkout_session`` was pulled from ``scheduling_service``, 

41 which has never defined it — an ImportError that made this whole module 

42 unimportable. It lives in ``attendance_service``. 

43 * ``datetime.now(timezone.utc)`` (aware) was compared against 

44 ``datetime.combine(date, time)`` (always naive), so the first expired 

45 session raised ``TypeError: can't compare offset-naive and offset-aware 

46 datetimes`` and the sweep died there. 

47 

48 On the comparison: ``Session.date`` and ``Session.end_time`` are stored as 

49 local wall-clock values, not UTC — the frontend reads them back through 

50 ``lib/sessionTime.ts`` and formats them as local time. So the comparison 

51 here is local-against-local on purpose. Do not "fix" this to UTC without 

52 also migrating the stored columns. 

53 """ 

54 now = datetime.now() 

55 from app.models.academy import AcademySettings 

56 

57 academies = Academy.query.all() 

58 total_checkout_count = 0 

59 

60 for academy in academies: 

61 settings = AcademySettings.query.filter_by(academy_id=academy.id).first() 

62 if not settings or not settings.auto_checkout_enabled: 

63 continue 

64 

65 # IN_PROGRESS only. A ``scheduled`` class has nobody checked in to 

66 # check out, and a class that was never started is the desk's to 

67 # cancel — not this job's. 

68 expired_sessions = Session.query.filter( 

69 Session.academy_id == academy.id, 

70 Session.status == "in_progress", 

71 ).all() 

72 

73 for session in expired_sessions: 

74 session_end = datetime.combine(session.date, session.end_time) 

75 if now > session_end: 

76 count = attendance_service.auto_checkout_session(session.id, academy.id) 

77 if count > 0: 

78 total_checkout_count += count 

79 logger.info( 

80 "Auto checkout: session %s — %s students", session.id, count 

81 ) 

82 

83 if total_checkout_count > 0: 

84 db.session.commit() 

85 logger.info( 

86 "Total auto checkouts across all academies: %s", total_checkout_count 

87 )

90def close_past_temporary_sessions(): 

91 """ 

92 Close out one-off classes whose day passed without ever being started. 

93 

94 The rule and the reasoning live in 

95 ``session_lifecycle_service.close_past_temporary_sessions`` — this is only 

96 the schedule and the log line. Kept thin on purpose: the lifecycle service 

97 owns session status, and a second copy of the rule here is how the two 

98 would drift apart. 

99 """ 

100 from app.services.session_lifecycle_service import ( 

101 close_past_temporary_sessions as _close, 

102 ) 

103 

104 closed = _close() 

105 if closed: 

106 logger.info("Closed %s unstarted one-off session(s)", closed)

109def check_overdue_payments():

114 academies = Academy.query.all() 

115 total_overdue_count = 0

149def check_upcoming_renewals(): 

150 """ 

151 Notify staff about upcoming billing renewals (due in 3 days). 

152 """ 

153 from datetime import date, timedelta 

154 from app.models.notification import Notification 

155 import uuid 

156 

157 academies = Academy.query.all() 

158 reminder_date = date.today() + timedelta(days=3) 

159 

160 for academy in academies: 

161 from app.models.academy import AcademySettings 

162 settings = AcademySettings.query.filter_by(academy_id=academy.id).first() 

163 days_before = settings.billing_reminder_days_before if settings else 3 

164 

165 target_date = date.today() + timedelta(days=days_before) 

166 

167 upcoming = ( 

168 StudentBilling.query.join(StudentBilling.student) 

169 .filter( 

170 StudentBilling.status == "due", 

171 StudentBilling.due_date == target_date, 

172 ) 

173 .all() 

174 ) 

175 

176 for billing in upcoming: 

177 student = billing.student 

178 student_name = f"{student.first_name} {student.last_name}" if student else "Unknown" 

179 

180 notification = Notification( 

181 id=str(uuid.uuid4()), 

182 academy_id=academy.id, 

183 user_id=None, # Broadcast 

184 type="payment_reminder", 

185 title=f"Payment due in {days_before} days", 

186 message=f"Payment of {billing.amount_da} DA due for {student_name}", 

187 detail=f"Plan: {billing.payment_plan.name if billing.payment_plan else 'N/A'}", 

188 ) 

189 db.session.add(notification) 

190 

191 db.session.commit()
