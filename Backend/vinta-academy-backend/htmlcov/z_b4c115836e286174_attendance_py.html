1""" 

2Attendance schemas — Check-in/out, Roster, Auto-checkout.

8# ── Request Schemas ──────────────────────────────────────────────── 

9 

10class CheckInRequestSchema(Schema): 

11 """POST /api/attendance/check-in""" 

12 session_id = fields.String(required=True, metadata={"description": "Session ID", "example": "uuid-string"}) 

13 student_id = fields.String(required=True, metadata={"description": "Student ID", "example": "uuid-string"}) 

14 pin = fields.String(required=True, metadata={"description": "Staff PIN for attribution", "example": "1234"}) 

15 status = fields.String(load_default="PRESENT", metadata={"description": "PRESENT or ABSENT", "example": "PRESENT"}) 

16 is_group_swap = fields.Boolean(load_default=False, metadata={"description": "True when student swaps in from another group"})

73class RosterEntrySchema(Schema): 

74 """Single entry in the session roster.""" 

75 id = fields.String(metadata={"description": "SessionStudent record ID"}) 

76 student_id = fields.String(metadata={"description": "Student ID"}) 

77 student_name = fields.String(metadata={"description": "Student full name"}) 

78 is_present = fields.Boolean(metadata={"description": "Attendance status"}) 

79 status = fields.String(metadata={"description": "PRESENT | ABSENT, mirrors is_present"}) 

80 is_group_swap = fields.Boolean( 

81 metadata={"description": "True when this attendance counts for another group (swap/makeup)"} 

82 ) 

83 checked_in_at = fields.String(metadata={"description": "Check-in timestamp (ISO 8601)"}) 

84 checked_out_at = fields.String(metadata={"description": "Check-out timestamp (ISO 8601)"}) 

85 checked_in_by = fields.String(metadata={"description": "User ID who performed check-in"}) 

86 timestamp = fields.String( 

87 metadata={"description": "When attendance was recorded; null until someone acts"} 

88 ) 

89 # NOTE: there is deliberately no payment_status here. Whether a student has 

90 # paid is owned by their subscription and reported via remaining_credits / 

91 # access_end / badges below — a second copy on this row could only drift. 

92 remaining_credits = fields.Integer(metadata={"description": "Remaining credits (if tracked)"}) 

93 access_end = fields.String(metadata={"description": "Access end date (ISO 8601)"}) 

94 badges = fields.List(fields.String, metadata={"description": "RENEW_REQUIRED / ATTENDANCE_WARNING badges"})
