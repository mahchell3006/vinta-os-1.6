20@analytics_bp.route("/dashboard", methods=["GET"]) 

21@jwt_required() 

22@tenant_required 

23def get_dashboard(): 

24 """ 

25 Dashboard overview stats. 

26 Returns: total students, total teachers, total classes, 

27 today's sessions, weekly agenda summary. 

28 """ 

29 from flask import g 

30 from datetime import date, timedelta 

31 from app.models.student import Student 

32 from app.models.teacher import Teacher 

33 from app.models.class_room import Class 

34 from app.models.scheduling import Session 

35 

36 academy_id = g.current_academy_id 

37 

38 total_students = Student.query.filter_by(academy_id=academy_id).count() 

39 total_teachers = Teacher.query.filter_by(academy_id=academy_id).count() 

40 total_classes = Class.query.filter_by(academy_id=academy_id).count() 

41 

42 # Today's sessions 

43 today = date.today() 

44 today_sessions = Session.query.filter( 

45 Session.academy_id == academy_id, 

46 Session.date == today, 

47 ).count() 

48 

49 # This week's sessions count 

50 week_start = today - timedelta(days=today.weekday() + 1) 

51 week_end = week_start + timedelta(days=6) 

52 week_sessions = Session.query.filter( 

53 Session.academy_id == academy_id, 

54 Session.date >= week_start, 

55 Session.date <= week_end, 

56 ).count() 

57 

58 # Billing stats 

59 billing_stats = billing_service.get_billing_stats(academy_id) 

60 

61 # Monthly income 

62 from sqlalchemy import func 

63 from app.models.billing import StudentBilling 

64 from calendar import monthrange 

65 month_start = today.replace(day=1) 

66 _, last_day = monthrange(today.year, today.month) 

67 month_end = month_start.replace(day=last_day) 

68 monthly_income = ( 

69 db.session.query(func.sum(StudentBilling.amount_da)) 

70 .join(Student) 

71 .filter( 

72 Student.academy_id == academy_id, 

73 StudentBilling.status == "paid", 

74 StudentBilling.paid_date >= month_start, 

75 StudentBilling.paid_date <= month_end, 

76 ) 

77 .scalar() or 0 

78 ) 

79 

80 return jsonify({ 

81 "total_students": total_students, 

82 "total_teachers": total_teachers, 

83 "total_classes": total_classes, 

84 "today_sessions": today_sessions, 

85 "week_sessions": week_sessions, 

86 "monthly_income": int(monthly_income), 

87 "billing": billing_stats, 

88 }), 200
